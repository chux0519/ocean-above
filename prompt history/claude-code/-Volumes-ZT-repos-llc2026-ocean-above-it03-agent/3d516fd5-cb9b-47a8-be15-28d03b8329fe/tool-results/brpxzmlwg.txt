"""One lock per conversation, shared by chat turns and the booking endpoint (design §4.6, B7).

In-process only: the API runs as a single uvicorn worker. The database does not rely on this lock
for B5/B7 (those hold through `BEGIN IMMEDIATE` and `tickets.quote_id UNIQUE`); the lock keeps a
booking click and a running chat turn of the same conversation from interleaving.

The lock is NOT reentrant. `Airline.book` takes it itself; everything else in `Airline` expects
the caller (the chat turn) to hold it.
"""
from __future__ import annotations

import asyncio
import threading
import time
from contextlib import asynccontextmanager, contextmanager
from typing import AsyncIterator, Iterator


class ConversationBusy(Exception):
    """The conversation lock was not obtained within the timeout."""


class ConversationLocks:
    def __init__(self) -> None:
        self._mutex = threading.Lock()
        self._entries: dict[str, list] = {}  # conversation id -> [lock, holders + waiters]

    def _checkout(self, conversation_id: str) -> threading.Lock:
        with self._mutex:
            entry = self._entries.setdefault(conversation_id, [threading.Lock(), 0])
            entry[1] += 1
            return entry[0]

    def _checkin(self, conversation_id: str) -> None:
        with self._mutex:
            entry = self._entries[conversation_id]
            entry[1] -= 1
            if entry[1] == 0:
                del self._entries[conversation_id]

    @contextmanager
    def hold(self, conversation_id: str, timeout: float | None = None) -> Iterator[None]:
        """Blocking form, for sync code (threadpool endpoints). Never call it on the event loop:
        use `hold_async` there."""
        lock = self._checkout(conversation_id)
        try:
            if not lock.acquire(timeout=-1 if timeout is None else timeout):
                raise ConversationBusy(conversation_id)
            try:
                yield
            finally:
                lock.release()
        finally:
            self._checkin(conversation_id)

    @asynccontextmanager
    async def hold_async(self, conversation_id: str, timeout: float | None = None,
                         poll: float = 0.01) -> AsyncIterator[None]:
        """Async form of the same lock: waits without blocking the event loop, safe to cancel
        while waiting, and may be held across awaits (a whole streamed turn)."""
        lock = self._checkout(conversation_id)
        try:
            deadline = None if timeout is None else time.monotonic() + timeout
            while not lock.acquire(blocking=False):
                if deadline is not None and time.monotonic() >= deadline:
                    raise ConversationBusy(conversation_id)
                await asyncio.sleep(poll)
            try:
                yield
            finally:
                lock.release()
        finally:
            self._checkin(conversation_id)

    def locked(self, conversation_id: str) -> bool:
        with self._mutex:
            entry = self._entries.get(conversation_id)
            return entry is not None and entry[0].locked()

def _row(row: sqlite3.Row | None, json_cols: Iterable[str] = ()) -> dict | None:
    if row is None:
        return None
    out = dict(row)
    for col in json_cols:
        if out[col] is not None:
            out[col] = json.loads(out[col])
    return out

def upsert_user(con, now: str, auth_provider: str, provider_sub: str, display_name: str,
                email: str | None = None) -> dict:
    """The user for (auth_provider, provider_sub); created on first sight, name/email refreshed."""
    con.execute(
        "INSERT INTO users (id, auth_provider, provider_sub, email, display_name, created_at) "
        "VALUES (?, ?, ?, ?, ?, ?) "
        "ON CONFLICT (auth_provider, provider_sub) DO UPDATE SET "
        "email = excluded.email, display_name = excluded.display_name",
        (new_id("u"), auth_provider, provider_sub, email, display_name, now),
    )
    row = con.execute(
        f"SELECT {_USER_COLS} FROM users WHERE auth_provider = ? AND provider_sub = ?",
        (auth_provider, provider_sub),
    ).fetchone()
    return _row(row)


def get_user(con, user_id: str) -> dict | None:
    return _row(con.execute(f"SELECT {_USER_COLS} FROM users WHERE id = ?", (user_id,)).fetchone())

def set_quote_status(con, quote: dict, status: str, now: str, reason: str | None = None) -> None:
    """Move a quote to `status` and update the dict in place. `presented` stamps presented_at;
    the terminal states stamp closed_at. The row must still hold the status the caller read."""
    column = "presented_at" if status == "presented" else "closed_at"
    cur = con.execute(
        f"UPDATE quotes SET status = ?, status_reason = ?, {column} = ? WHERE id = ? AND status = ?",
        (status, reason, now, quote["id"], quote["status"]),
    )
    if cur.rowcount != 1:
        raise RuntimeError(f"quote {quote['id']} changed underneath us")
    quote.update({"status": status, "status_reason": reason, column: now})

def mark_nonce_used(con, quote_id: str, message_id: str, now: str) -> bool:
    """True when this call consumed the nonce; False when it had been used before."""
    cur = con.execute(
        "UPDATE quote_cards SET nonce_used_at = ? "
        "WHERE quote_id = ? AND message_id = ? AND nonce_used_at IS NULL",
        (now, quote_id, message_id),
    )
    return cur.rowcount == 1

def make_db(path, locations, links=()):
    """A sqlite file with the two tables but without constraints, so bad rows can be stored."""
    con = sqlite3.connect(path)
    con.execute("create table locations (city_name, coordinate, is_hub, connected_hubs)")
    con.execute("create table hub_links (hub_a, hub_b)")
    con.executemany("insert into locations values (?,?,?,?)", locations)
    con.executemany("insert into hub_links values (?,?)", links)
    con.commit()
    con.close()
    return path

def create_app(db_path: str | Path | None = None) -> FastAPI:
    path = Path(db_path or os.environ.get("LOCATIONS_DB") or DEFAULT_DB)

    @asynccontextmanager
    async def lifespan(app: FastAPI):
        app.state.networks = NetworkProvider(path)
        app.state.networks.current()  # fail at startup, not on the first request, if the db is bad
        yield

    app = FastAPI(title="Ocean Above API", lifespan=lifespan)

    @app.get("/health")
    def health(request: Request) -> dict:
        net = request.app.state.networks.current()
        return {
            "status": "ok",
            "network": {
                "cities": len(net.cities),
                "hubs": len(net.hubs),
                "hub_links": len(net.hub_links),
                "unserved": len(net.cities) - len(net.served),
                "fingerprint": net.fingerprint.id,
            },
        }

    @app.get("/api/cities")
    def cities(request: Request) -> dict:
        """Debug listing straight from the loaded network (database order)."""
        net = request.app.state.networks.current()
        return {
            "fingerprint": net.fingerprint.id,
            "count": len(net.cities),
            "cities": [
                {
                    "name": c.name,
                    "coordinate": [str(c.x), str(c.y)],
                    "is_hub": c.is_hub,
                    "served": net.is_served(c.name),
                    "connected_hubs": list(c.connected_hubs),
                }
                for c in net.cities.values()
            ],
        }

    return app


app = create_app()

#!/usr/bin/env python3
"""Rule-registry lint. `python3 tool/review_lint.py` runs all rules; `... R01` runs one.

Add a rule: write a function returning a list of problems, decorate it with @rule('Rxx', 'what it checks').
Keep each rule tied to an incident in docs/review_rules.md.
"""
import re
import subprocess
import sys
from pathlib import Path

ROOT = Path(__file__).resolve().parents[1]
RULES = {}


def rule(rid, title):
    def wrap(fn):
        RULES[rid] = (title, fn)
        return fn
    return wrap


def tracked(*globs):
    out = subprocess.run(['git', 'ls-files', '--', *globs], cwd=ROOT, capture_output=True, text=True).stdout
    return [ROOT / p for p in out.splitlines() if p]


# Fill in the names you must never ship (machine names, user names, internal domains).
FORBIDDEN_NAMES = [r'\bDESKTOP-[A-Z0-9]{5,}\b', r'(?i)yongsheng', r'(?i)\bmacmini\b']
# Raw agent sessions must stay unedited (deliverable requirement); this file lists the patterns itself.
NAME_SCAN_EXCLUDE = ('prompt history/', 'tool/review_lint.py')


@rule('R01', 'No real host or user names in tracked files')
def no_real_names():
    problems = []
    for f in tracked():
        if not f.is_file() or str(f.relative_to(ROOT)).startswith(NAME_SCAN_EXCLUDE):
            continue
        text = f.read_text(errors='ignore')
        for pat in FORBIDDEN_NAMES:
            for m in re.finditer(pat, text):
                problems.append(f'{f.relative_to(ROOT)}: {m.group(0)}')
    return problems


@rule('R02', 'rm -rf only as rm -rf "${VAR:?}"')
def safe_rm():
    problems = []
    for f in tracked('*.sh', '**/*.sh'):
        for n, line in enumerate(f.read_text(errors='ignore').splitlines(), 1):
            if re.search(r'rm\s+-rf\s+"?\$(?!\{\w+:\?\})', line):
                problems.append(f'{f.relative_to(ROOT)}:{n}: {line.strip()}')
    return problems


def main(argv):
    selected = argv or sorted(RULES)
    failed = False
    for rid in selected:
        title, fn = RULES[rid]
        problems = fn()
        print(f'{rid} {title}: ' + ('ok' if not problems else f'{len(problems)} problem(s)'))
        for p in problems:
            print('   ', p)
        failed |= bool(problems)
    return 1 if failed else 0


if __name__ == '__main__':
    sys.exit(main(sys.argv[1:]))
