"""LLM settings from the environment. The key is read here and nowhere else; it is never logged
and never part of a repr."""
from __future__ import annotations

import os
from dataclasses import dataclass, field
from pathlib import Path
from typing import Mapping

DEFAULT_BASE_URL = "https://api.deepseek.com"
# DeepSeek's flash model id on 2026-11-23 (GET /models: "deepseek-flash" = DeepSeek-V4.1-Flash).
DEFAULT_MODEL = "deepseek-flash"
REPO_ROOT = Path(__file__).resolve().parents[3]
EFFORTS = ("none", "low", "high", "max")  # what a request may ask for


class LLMNotConfigured(Exception):
    """No API key in the environment or the env file."""


def read_env_file(path: str | Path) -> dict[str, str]:
    """KEY=VALUE lines of a .env file (comments, blank lines and `export ` are skipped)."""
    out: dict[str, str] = {}
    try:
        text = Path(path).read_text(encoding="utf-8")
    except OSError:
        return out
    for line in text.splitlines():
        line = line.strip()
        if not line or line.startswith("#") or "=" not in line:
            continue
        key, value = line.removeprefix("export ").split("=", 1)
        value = value.strip()
        if len(value) >= 2 and value[0] == value[-1] and value[0] in "'\"":
            value = value[1:-1]
        out[key.strip()] = value
    return out


@dataclass(frozen=True)
class LLMConfig:
    api_key: str = field(repr=False)
    base_url: str = DEFAULT_BASE_URL
    model: str = DEFAULT_MODEL
    # Thinking level (DeepSeek: `thinking` + `reasoning_effort`, docs "Thinking Mode"):
    #   "low" | "high" | "max" -> {"thinking": {"type": "enabled"}, "reasoning_effort": <level>}
    #   "none"                 -> {"thinking": {"type": "disabled"}}
    #   "omit"                 -> neither field, for OpenAI-compatible servers that reject them
    reasoning_effort: str = "high"

    def __post_init__(self) -> None:
        if self.reasoning_effort not in (*EFFORTS, "omit"):
            raise ValueError(f"LLM_REASONING_EFFORT must be one of {', '.join(EFFORTS)} or omit")

    @classmethod
    def from_env(cls, env: Mapping[str, str] | None = None) -> "LLMConfig":
        """`LLM_API_KEY` (fallback `DEEPSEEK_API_KEY`), `LLM_BASE_URL`, `LLM_MODEL`,
        `LLM_REASONING_EFFORT` (default high). Process environment first; then the file named by `OCEAN_ENV_FILE`
        (default: `.env` in the repository root), so a worktree can point at the main checkout."""
        env = os.environ if env is None else env
        file_values = read_env_file(env.get("OCEAN_ENV_FILE") or REPO_ROOT / ".env")

        def get(name: str) -> str:
            return (env.get(name) or file_values.get(name) or "").strip()

        key = get("LLM_API_KEY") or get("DEEPSEEK_API_KEY")
        if not key:
            raise LLMNotConfigured("set LLM_API_KEY (or DEEPSEEK_API_KEY), or OCEAN_ENV_FILE")
        return cls(
            api_key=key,
            base_url=(get("LLM_BASE_URL") or DEFAULT_BASE_URL).rstrip("/"),
            model=get("LLM_MODEL") or DEFAULT_MODEL,
            reasoning_effort=(get("LLM_REASONING_EFFORT") or "high").lower(),
        )
"""Streaming client for an OpenAI-compatible chat completions API (plain httpx, no SDK).

`stream()` yields `TextDelta`s as the model writes and exactly one `Completion` at the end, with
the tool calls assembled from their fragments. Any failure is an `LLMError` that says whether a
retry makes sense. The API key only ever goes into the Authorization header.
"""
from __future__ import annotations

import json
import re
from dataclasses import dataclass, field
from typing import Any, AsyncIterator, Protocol

import httpx

from .budget import AgentBudget
from .config import EFFORTS, LLMConfig

MAX_TOKENS = AgentBudget().max_output_tokens_per_call  # includes the reasoning tokens of thinking mode


@dataclass(frozen=True)
class TextDelta:
    text: str


@dataclass
class Completion:
    content: str = ""
    tool_calls: list[dict] = field(default_factory=list)  # [{"id", "name", "arguments": str}]
    finish_reason: str | None = None
    usage: dict | None = None
    reasoning: str = ""

    def assistant_message(self) -> dict:
        """The message to append to the context before the tool results (tool calls in their
        wire-safe form, see `wire_tool_call`)."""
        message: dict[str, Any] = {"role": "assistant", "content": self.content or ""}
        if self.reasoning:
            message["reasoning_content"] = self.reasoning
        if self.tool_calls:
            message["tool_calls"] = [wire_tool_call(c["id"], c["name"], c["arguments"]) for c in self.tool_calls]
        return message

    def as_record(self) -> dict:
        """What is stored in llm_calls: the reasoning is kept verbatim for provenance and to pass
        it back in later requests; it is never shown to the user and never guarded as reply text."""
        return {"content": self.content, "reasoning_content": self.reasoning,
                "tool_calls": self.tool_calls, "finish_reason": self.finish_reason, "usage": self.usage}

    @property
    def total_tokens(self) -> int | None:
        return (self.usage or {}).get("total_tokens")

    @property
    def output_chars(self) -> int:
        """Characters the model produced: text, reasoning, tool call names and arguments."""
        return len(self.content) + len(self.reasoning) + sum(
            len(str(c.get("name") or "")) + len(str(c.get("arguments") or "")) for c in self.tool_calls)


_TOOL_NAME = re.compile(r"^[A-Za-z0-9_-]{1,64}$")
INVALID_TOOL_NAME = "invalid_tool"


def wire_tool_call(call_id: str, name: str, arguments: Any) -> dict:
    """A tool call as it is sent back to the LLM. A name the API may reject (OpenAI:
    `^[a-zA-Z0-9_-]{1,64}$`) becomes `invalid_tool`, and arguments that are not a JSON object
    become `{}`, so one broken call cannot make every later request of the conversation fail.
    The raw values stay in `tool_calls` (app.db) and in the tool result's error."""
    if not isinstance(name, str) or not _TOOL_NAME.match(name):
        name = INVALID_TOOL_NAME
    if not isinstance(arguments, str):
        arguments = json.dumps(arguments, ensure_ascii=False, separators=(",", ":"))
    try:
        ok = isinstance(json.loads(arguments), dict)
    except (ValueError, RecursionError):
        ok = False
    return {"id": call_id, "type": "function",
            "function": {"name": name, "arguments": arguments if ok else "{}"}}


class _ThinkSplitter:
    """Moves a `<think>...</think>` block at the start of `content` (servers that inline the
    reasoning instead of using `reasoning_content`) out of the visible text."""

    OPEN, CLOSE = "<think>", "</think>"

    def __init__(self) -> None:
        self.state = "start"  # start -> think -> text
        self.buf = ""

    def feed(self, text: str) -> tuple[str, str]:
        """(visible text, reasoning text) for this chunk."""
        if self.state == "text":
            return text, ""
        if self.state == "lead":  # white space right after </think> is dropped
            text = text.lstrip()
            if text:
                self.state = "text"
            return text, ""
        self.buf += text
        if self.state == "start":
            head = self.buf.lstrip()
            if self.OPEN.startswith(head):  # may still become "<think>"
                return "", ""
            if not head.startswith(self.OPEN):
                self.state, out, self.buf = "text", self.buf, ""
                return out, ""
            self.state, self.buf = "think", head[len(self.OPEN):]
        end = self.buf.find(self.CLOSE)
        if end < 0:  # keep a possible partial "</think>" at the end
            keep = max((k for k in range(1, len(self.CLOSE)) if self.buf.endswith(self.CLOSE[:k])), default=0)
            reasoning, self.buf = self.buf[:len(self.buf) - keep], self.buf[len(self.buf) - keep:]
            return "", reasoning
        reasoning, rest = self.buf[:end], self.buf[end + len(self.CLOSE):].lstrip()
        self.state, self.buf = ("text" if rest else "lead"), ""
        return rest, reasoning

    def finish(self) -> tuple[str, str]:
        rest, self.buf = self.buf, ""
        if self.state == "think":
            return "", rest
        return rest, ""


class LLMError(Exception):
    """A failed LLM call. `received` says what it may have cost (06 review P1-2): None when no
    response body was read (HTTP error status, connection / TLS failure, timeout before the
    response): the provider generated nothing. Otherwise the number of output characters
    (text, reasoning, tool call names and arguments) received before the stream broke. `usage`:
    the usage block, if the provider sent one before the failure."""

    def __init__(self, message: str, *, retryable: bool, status: int | None = None,
                 received: int | None = None, usage: dict | None = None):
        super().__init__(message)
        self.retryable = retryable
        self.status = status
        self.received = received
        self.usage = usage

class OpenAIChatClient:
    """One instance per event loop (it keeps an httpx.AsyncClient); `await aclose()` at shutdown."""

    def __init__(self, config: LLMConfig, *, transport: httpx.AsyncBaseTransport | None = None,
                 timeout: httpx.Timeout | None = None, temperature: float = 0,
                 max_tokens: int = MAX_TOKENS):
        self.config = config
        self.model = config.model
        self.reasoning_effort = config.reasoning_effort
        self.temperature = temperature
        self.max_tokens = max_tokens
        self._transport = transport
        self._timeout = timeout or httpx.Timeout(connect=10, read=60, write=30, pool=10)
        self._client: httpx.AsyncClient | None = None

    def _http(self) -> httpx.AsyncClient:
        if self._client is None:
            self._client = httpx.AsyncClient(transport=self._transport, timeout=self._timeout)
        return self._client

    async def aclose(self) -> None:
        if self._client is not None:
            await self._client.aclose()
            self._client = None

    def body(self, messages: list[dict], tools: list[dict], reasoning_effort: str | None = None) -> dict:
        body: dict[str, Any] = {
            "model": self.model,
            "messages": messages,
            "temperature": self.temperature,
            "max_tokens": self.max_tokens,
            "stream": True,
            "stream_options": {"include_usage": True},
        }
        if tools:
            body["tools"] = tools
            body["tool_choice"] = "auto"
        effort = reasoning_effort or self.reasoning_effort
        if reasoning_effort is not None and reasoning_effort not in EFFORTS:
            raise ValueError(f"reasoning_effort must be one of {', '.join(EFFORTS)}")
        if effort == "none":
            body["thinking"] = {"type": "disabled"}
        elif effort != "omit":
            # thinking mode ignores `temperature` (DeepSeek docs); it is sent anyway, harmlessly
            body["thinking"] = {"type": "enabled"}
            body["reasoning_effort"] = effort
        return body

    def _scrub(self, text: str) -> str:
        return text.replace(self.config.api_key, "***") if self.config.api_key else text

    async def stream(self, messages: list[dict], tools: list[dict], *,
                     reasoning_effort: str | None = None) -> AsyncIterator[TextDelta | Completion]:
        body = self.body(messages, tools, reasoning_effort)
        done = Completion()
        think = _ThinkSplitter()
        calls: dict[int, dict] = {}
        headers = {"Authorization": f"Bearer {self.config.api_key}", "Accept": "text/event-stream"}
        received: int | None = None  # output characters, once a 200 response is being read
        try:
            async with self._http().stream("POST", f"{self.config.base_url}/chat/completions",
                                           json=body, headers=headers) as response:
                if response.status_code != 200:
                    status = response.status_code
                    detail = ""
                    if status not in (401, 403):  # an auth error body may quote the key
                        detail = ": " + self._scrub((await response.aread()).decode("utf-8", "replace"))[:300]
                    raise LLMError(f"LLM HTTP {status}{detail}", status=status,
                                   retryable=status in (408, 409, 429) or status >= 500)
                received = 0
                async for line in response.aiter_lines():
                    if not line.startswith("data:"):
                        continue  # blank separators, `: keep-alive` comments
                    data = line[5:].strip()
                    if data == "[DONE]":
                        break
                    try:
                        chunk = json.loads(data)
                    except ValueError:
                        raise LLMError("LLM sent a chunk that is not JSON", retryable=True) from None
                    if chunk.get("error"):
                        raise LLMError("LLM stream error: " + self._scrub(json.dumps(chunk["error"]))[:300],
                                       retryable=True)
                    if chunk.get("usage"):
                        done.usage = chunk["usage"]
                    for choice in chunk.get("choices") or []:
                        if choice.get("index", 0) != 0:
                            continue
                        delta = choice.get("delta") or {}
                        if delta.get("reasoning_content"):  # thinking: collected, never yielded
                            done.reasoning += delta["reasoning_content"]
                            received += len(delta["reasoning_content"])
                        if delta.get("content"):
                            received += len(delta["content"])
                            visible, thought = think.feed(delta["content"])
                            done.reasoning += thought
                            if visible:
                                done.content += visible
                                yield TextDelta(visible)
                        for part in delta.get("tool_calls") or []:
                            call = calls.setdefault(part.get("index", 0), {"id": "", "name": "", "arguments": ""})
                            function = part.get("function") or {}
                            received += len(function.get("name") or "") + len(function.get("arguments") or "")
                            call["id"] = part.get("id") or call["id"]
                            call["name"] += function.get("name") or ""
                            call["arguments"] += function.get("arguments") or ""
                        if choice.get("finish_reason"):
                            done.finish_reason = choice["finish_reason"]
        except LLMError as exc:
            exc.received, exc.usage = received, done.usage
            raise
        except (httpx.HTTPError, httpx.StreamError) as exc:  # timeouts, connections, cut streams
            raise LLMError(f"LLM transport error: {type(exc).__name__}", retryable=True,
                           received=received, usage=done.usage) from None
        except Exception as exc:  # a chunk of an unexpected shape, undecodable bytes, ...
            raise LLMError(f"LLM stream could not be read: {type(exc).__name__}", retryable=True,
                           received=received, usage=done.usage) from None
        visible, thought = think.finish()
        done.reasoning += thought
        if visible:
            done.content += visible
            yield TextDelta(visible)
        if done.finish_reason is None:
            raise LLMError("LLM stream ended without a finish_reason", retryable=True,
                           received=received, usage=done.usage)
        for index in sorted(calls):
            call = calls[index]
            call["id"] = call["id"] or f"call_{index}"
            done.tool_calls.append(call)
        yield done
"""The agent loop (design §4): one chat turn as a stream of contract SSE events.

    user message stored -> [LLM -> tools -> cards]* -> guarded text -> assistant message stored

The LLM chooses tools and writes text; routes, prices and tickets come from the airline through
the four tools. Text is released sentence by sentence after the ReplyGuard traced every number
and city in it; the first problem replaces the whole reply with a code template. The turn's
quote cards become bookable only when `Airline.post_assistant_message` stores them (`done`).

Nothing here can book: the loop talks to the airline through `Toolbox` (four tools, no booking)
and `post_assistant_message`. Booking is the HTTP endpoint's call, caused by the user's click.
"""

class Agent:
    def __init__(self, airline: Any, llm: LLM, *, budget: AgentBudget | None = None,
                 max_steps: int | None = None,
                 ping_interval: float = 15.0, retries: int = 2, backoff: float = 0.5,
                 llm_timeout: float = 60.0, turn_timeout: float | None = None,
                 lock_timeout: float | None = LOCK_TIMEOUT,
                 concurrency: TurnConcurrency | None = None, drain_timeout: float = DRAIN_TIMEOUT,
                 sleep: Callable[[float], Awaitable[None]] = asyncio.sleep):
        """`budget`: the per-turn limits (design §4.8; default `AgentBudget()`). `max_steps` and
        `turn_timeout` override its fields (tests, older callers). `concurrency`: the server-wide
        limit on running turns (design §4.9; default `TurnConcurrency()`). `drain_timeout`: how
        long a cut-short turn waits for its worker threads before it ends anyway."""
        budget = budget or DEFAULT_BUDGET
        if max_steps is not None:
            budget = replace(budget, max_steps=max_steps)
        if turn_timeout is not None:
            budget = replace(budget, max_turn_seconds=turn_timeout)
        self.budget = budget
        self.llm = llm
        self.store = airline.store
        self.ping_interval = ping_interval
        self.retries = retries
        self.backoff = backoff
        self.llm_timeout = llm_timeout
        self.lock_timeout = lock_timeout
        self.slots = TurnSlots(concurrency or TurnConcurrency())
        self.drain_timeout = drain_timeout
        self._sleep = sleep
        self._tools = Toolbox(airline)
        # The only other things the loop takes from the airline. No reference to the Airline
        # object itself is kept.
        self._post_assistant_message = airline.post_assistant_message
        self._locks = airline.locks
        self._networks = airline.networks
        self._index: tuple[str, CityIndex] | None = None
        self._tasks: set[asyncio.Task] = set()
        self._user_turns: dict[str, int] = {}  # user id -> running turns (event loop only)

    @property
    def max_steps(self) -> int:
        return self.budget.max_steps

    @property
    def turn_timeout(self) -> float:
        return self.budget.max_turn_seconds

    # ------------------------------------------------------------------ public -------------
    def stream_turn(self, user_id: str, conversation_id: str, text: str, *,
                    reasoning_effort: str | None = None,
                    on_not_stored: Callable[[], Any] | None = None,
                    on_server_busy: Callable[[], Any] | None = None,
                    owner_checked: bool = False) -> AsyncIterator[Event]:
        """Start a turn and return its events: `user_message`, then `status` / `card` / `delta` /
        `replace` / `ping`, ending with exactly one `done` or `error`.

        `reasoning_effort` overrides the LLM's thinking level for this turn (none / low / high /
        max; None = configured default). `on_not_stored`: a blocking callable run (in a worker
        thread, before the event is sent) when the turn ends before the user's message was
        stored: refused with `server_busy`, or app.db failed to store the message
        (`internal_error`) - the API gives the day's message slot back there. It runs even when
        nobody reads the stream. `on_server_busy` is its older name.
        `owner_checked`: the caller already checked (off the event loop) that the conversation is
        the user's, so this method reads nothing from app.db (06 review P2-9); the turn's own
        writes still check ownership.

        Raises before anything is stored: NotFound (not the user's conversation), ValueError
        (empty or too long text, unknown reasoning_effort). Must be called on the event loop. The turn runs in its own
        task: it finishes and is stored even when the client stops reading."""
        if not isinstance(text, str) or not text.strip():
            raise ValueError("text must not be empty")
        text = text.strip()
        if len(text) > MAX_USER_CHARS:
            raise ValueError(f"text is longer than {MAX_USER_CHARS} characters")
        if reasoning_effort is not None and reasoning_effort not in EFFORTS:
            raise ValueError(f"reasoning_effort must be one of {', '.join(EFFORTS)}")
        if not owner_checked and self.store.get_conversation(user_id, conversation_id) is None:
            raise NotFound(conversation_id)
        queue: asyncio.Queue[Event | None] = asyncio.Queue()
        task = asyncio.get_running_loop().create_task(
            self._run(user_id, conversation_id, text, reasoning_effort, queue, on_not_stored or on_server_busy))
        self._tasks.add(task)
        task.add_done_callback(self._tasks.discard)
        return self._events(queue)

    async def wait_idle(self) -> None:
        """Wait for every running turn (shutdown, tests)."""
        while pending := [t for t in self._tasks if not t.done()]:
            await asyncio.wait(pending)

    async def _events(self, queue: asyncio.Queue) -> AsyncIterator[Event]:
        while True:
            try:
                event = await asyncio.wait_for(queue.get(), self.ping_interval)
            except asyncio.TimeoutError:
                yield Event("ping", {})
                continue
            if event is None:
                return
            yield event

    # ------------------------------------------------------------------ one turn -----------
    async def _run(self, user_id: str, conversation_id: str, text: str, reasoning_effort: str | None,
                   queue: asyncio.Queue, on_not_stored: Callable[[], Any] | None = None) -> None:
        def emit(name: str, data: dict) -> None:
            queue.put_nowait(Event(name, data))

        async def not_stored() -> None:
            if on_not_stored is not None:
                try:
                    await asyncio.to_thread(on_not_stored)
                except Exception:
                    log.exception("not-stored callback failed (conversation %s)", conversation_id)

        # The conversation lock first, then the user's own limit, then a server-wide slot: a
        # second message of a busy conversation waits without taking a slot. Every refusal comes
        # before anything is stored.
        try:
            async with self._locks.hold_async(conversation_id, timeout=self.lock_timeout):
                with self._user_turn(user_id):
                    async with self.slots.hold():
                        await self._turn(user_id, conversation_id, text, reasoning_effort, emit)
        except ConversationBusy:
            emit("error", {"code": "conversation_busy", "message": BUSY})
        except _UserBusy:
            log.warning("chat turn refused: user already has %d turns running (conversation %s)",
                        self.slots.config.max_turns_per_user, conversation_id)
            emit("error", {"code": "conversation_busy", "message": USER_BUSY})
        except ServerBusy:
            log.warning("chat turn refused: %d turns running, none ended within %gs (conversation %s)",
                        self.slots.active, self.slots.config.queue_seconds, conversation_id)
            await not_stored()
            emit("error", {"code": "server_busy", "message": SERVER_BUSY})
        except _NotStored:
            log.exception("chat turn: the user's message could not be stored (conversation %s)", conversation_id)
            await not_stored()
            emit("error", {"code": "internal_error", "message": NOT_STORED})
        except Exception:
            log.exception("chat turn failed (conversation %s)", conversation_id)
            emit("error", {"code": "internal_error", "message": INTERNAL})
        finally:
            queue.put_nowait(None)

    @contextmanager
    def _user_turn(self, user_id: str) -> Iterator[None]:
        """At most `max_turns_per_user` running turns per user (06 review P2-8). Event loop only."""
        if self._user_turns.get(user_id, 0) >= self.slots.config.max_turns_per_user:
            raise _UserBusy()
        self._user_turns[user_id] = self._user_turns.get(user_id, 0) + 1
        try:
            yield
        finally:
            self._user_turns[user_id] -= 1
            if not self._user_turns[user_id]:
                del self._user_turns[user_id]

    async def _turn(self, user_id: str, conversation_id: str, text: str, reasoning_effort: str | None,
                    emit: Callable[[str, dict], None]) -> None:
        store = self.store
        conversation = await asyncio.to_thread(store.get_conversation, user_id, conversation_id)
        try:
            user_message = await asyncio.to_thread(store.add_message, user_id, conversation_id, "user", text)
        except Exception as exc:  # app.db timeout, the conversation gone: nothing of the turn exists
            raise _NotStored() from exc
        emit("user_message", message_view(user_message))
        if conversation is not None and not conversation["title"]:
            await asyncio.to_thread(store.set_conversation_title, user_id, conversation_id,
                                    " ".join(text.split())[:TITLE_CHARS])

        context = await asyncio.to_thread(build_context, store, user_id, conversation_id)
        index = await asyncio.to_thread(self._city_index)
        today = store.clock().date()
        guard = ReplyGuard(index, records=context.records, user_text=text,
                           earlier_user_texts=context.user_texts[:-1], today=today)
        turn = _Turn(user_id, conversation_id, user_message, guard, emit, BudgetTracker(self.budget),
                     reasoning_effort)
        messages = [{"role": "system", "content": system_prompt(today)}, *context.messages]

        try:
            try:
                async with asyncio.timeout(self.turn_timeout):
                    fallback = await self._steps(turn, messages)
            finally:
                await self._drain(turn)
        except TimeoutError:
            fallback = [{"kind": "turn_timeout", "value": str(self.turn_timeout)}]
        except LLMError as exc:
            fallback = [{"kind": "llm_error", "value": str(exc)[:200]}]
        except Exception as exc:  # a bug or a store error in the middle of the turn
            if not turn.outcomes and not turn.text:
                raise
            log.exception("chat turn failed after producing output (conversation %s)", conversation_id)
            fallback = [{"kind": "internal_error", "value": type(exc).__name__}]
        if fallback is not None and fallback[0]["kind"] in ("max_steps", "token_budget", "tool_budget"):
            log.warning("chat turn stopped by its budget (conversation %s): %s %s", conversation_id,
                        fallback[0]["kind"], turn.budget.snapshot())
        if fallback is not None and fallback[0]["kind"] in ("llm_error", "turn_timeout") \
                and not turn.outcomes and not turn.text:
            emit("error", {"code": "llm_unavailable", "message": LLM_UNAVAILABLE})
            return
        await self._finish(turn, fallback)

    async def _steps(self, turn: _Turn, messages: list[dict]) -> list[dict] | None:
        """The LLM / tool steps of a turn. Returns None when the LLM's text can stay, otherwise
        the problems that make the template reply replace it.

        Budgets (design §4.8): a step starts only while steps and tokens are left; each tool call
        runs only while calls are left for the step and the turn. A call that is not admitted is
        never executed: it gets a `tool_budget_exceeded` result (stored like any tool call, so
        every tool call of the completion has its result and the stored history stays well-formed).
        Once the turn's calls are used up and a call was refused, the turn ends with the template."""
        while (stop := turn.budget.stop_before_step()) is None:
            try:
                completion = await self._llm_step(turn, messages)
            except _Stopped as stopped:  # guard failure, or no retry left within the token cap
                return stopped.problems
            if not completion.tool_calls:
                if completion.finish_reason != "stop":  # "length", "content_filter", ...
                    return [{"kind": "truncated", "value": str(completion.finish_reason)}]
                if not turn.text.strip():
                    return [{"kind": "empty_reply", "value": ""}]
                return None
            if _flooded(completion, self.budget):
                # far more calls than a step may run (06 review P2-3): none runs and none is
                # stored one by one; the logged completion is marked so the next context skips it
                refused = len(completion.tool_calls)
                turn.budget.refused_calls += refused
                log.warning("chat turn: a completion with %d tool calls; none run (conversation %s)",
                            refused, turn.conversation_id)
                return [{"kind": "tool_budget", "value": str(refused)}]
            messages.append(completion.assistant_message())
            turn_exhausted = False
            for call in completion.tool_calls:
                refused = turn.budget.admit_tool_call()
                if refused is None:
                    turn.emit("status", {"text": STATUS_TEXT.get(call["name"], DEFAULT_STATUS)})
                    outcome = await turn.threads.run(self._tools.run, turn.user_id, turn.conversation_id,
                                                     call["name"], call["arguments"])
                else:
                    turn_exhausted = turn_exhausted or refused == "per_turn"
                    payload = refusal(refused, self.budget)
                    outcome = Outcome(call["name"], str(call["arguments"])[:MAX_RAW_ARGUMENTS], payload["code"], payload,
                                      error=f"budget: {refused}")
                # The outcome and its cards belong to the turn as soon as the tool returned, before
                # the row is written: if the time limit lands on the write, the reply still
                # carries the card that the stored row (written while draining) describes (P2-5).
                turn.outcomes.append(outcome)
                if refused is None:  # a refusal is no record a reply may quote
                    turn.guard.add_record(outcome.payload)
                for card in outcome.cards:
                    turn.cards.append(card)
                    turn.emit("card", {"card": card})
                messages.append({"role": "tool", "tool_call_id": call["id"], "content": outcome.for_llm()})
                await turn.threads.run(
                    self.store.add_tool_call, turn.user_id, turn.user_message["id"], outcome.name,
                    outcome.args, outcome.payload, error=outcome.error, latency_ms=outcome.latency_ms)
            if turn_exhausted:
                return [{"kind": "tool_budget", "value": str(self.budget.max_tool_calls_per_turn)}]
        return [stop]

    async def _drain(self, turn: _Turn) -> None:
        """Wait for worker threads the time limit (or an error) left running. A thread cannot be
        stopped; until it ends it may still write (a quote, a tool call row), so the reply is not
        stored and the conversation lock and the slot are not released before it ends - unless it
        outlives `drain_timeout` (store timeouts make that unlikely), which is logged."""
        if not turn.threads.running:
            return
        log.warning("chat turn cut short with %d worker thread(s) running; waiting up to %gs (conversation %s)",
                    turn.threads.running, self.drain_timeout, turn.conversation_id)
        if not await turn.threads.drain(self.drain_timeout):
            log.error("chat turn: %d worker thread(s) still running after %gs; ending the turn anyway "
                      "(conversation %s)", turn.threads.running, self.drain_timeout, turn.conversation_id)

    async def _finish(self, turn: _Turn, fallback: list[dict] | None = None) -> None:
        """Persist the assistant message (the `done` moment: quote cards become bookable)."""
        text = turn.text.rstrip()
        if fallback is not None:
            text = fallback_reply(turn.outcomes)
            turn.emit("replace", {"text": text})
        elif CLICK_TO_BOOK not in text and any(
                o.name == "request_booking" and o.code == "ok" for o in turn.outcomes):
            # fixed by code, whatever the LLM wrote: the click is what books
            piece = ("\n\n" if text else "") + CLICK_TO_BOOK
            turn.emit("delta", {"text": piece})
            text += piece
        message = await asyncio.to_thread(
            self._post_assistant_message, turn.user_id, turn.conversation_id, text, turn.cards, fallback)
        turn.emit("done", {"message": message})

    # ------------------------------------------------------------------ LLM step -----------
    def _release(self, turn: _Turn, piece: str, first_of_step: bool) -> None:
        """Check the reply so far plus `piece`; send `piece` when nothing is wrong. Checking the
        whole text catches a number or a city name that spans two pieces (`St. | Louis`)."""
        if not turn.text:
            piece = piece.lstrip()
        elif first_of_step and not turn.text[-1].isspace() and not piece[:1].isspace():
            piece = "\n\n" + piece  # text of a later step starts a new paragraph
        problems = turn.guard.check(turn.text + piece)
        if problems:
            raise _GuardFailed(problems)
        turn.text += piece
        turn.emit("delta", {"text": piece})

    async def _llm_step(self, turn: _Turn, messages: list[dict]) -> Completion:
        """One LLM call with retries. Text is released as it streams. A failed call is retried
        only while none of its text has been released."""
        request = {"turn": turn.user_message["id"], "messages": list(messages),
                   "tools": [t["function"]["name"] for t in TOOL_SCHEMAS],
                   "reasoning_effort": turn.reasoning_effort or getattr(self.llm, "reasoning_effort", None)}
        for attempt in range(self.retries + 1):
            started = time.monotonic()
            buffer = SentenceBuffer()
            completion: Completion | None = None
            released = False
            content = blank = ""

            def log_call(response: dict, usage: Any = None, *, output_chars: int | None = None,
                         no_response: bool = False) -> asyncio.Future:
                # every attempt counts against the turn's tokens, also failed ones and retries
                charged, estimated = turn.budget.charge(usage, messages, output_chars=output_chars,
                                                        no_response=no_response)
                response = {**response, "budget": {"charged_tokens": charged, "estimated": estimated,
                                                   "turn_tokens": turn.budget.tokens}}
                # `tokens` = what the turn was charged, so the site's daily budget (§8.2) counts
                # estimated calls the same way (06 review P2-9); `response.usage` keeps the report
                return turn.threads.start(
                    self.store.add_llm_call, turn.user_id, turn.conversation_id, request, response,
                    latency_ms=int((time.monotonic() - started) * 1000), tokens=charged,
                    model=getattr(self.llm, "model", None))

            try:
                # aclosing: a guard failure or a timeout must close the HTTP stream right away
                async with asyncio.timeout(self.llm_timeout), \
                        aclosing(self.llm.stream(
                            messages, TOOL_SCHEMAS, reasoning_effort=turn.reasoning_effort)) as stream:
                    async for event in stream:
                        if isinstance(event, TextDelta):
                            content += event.text
                            for sentence in buffer.push(event.text):
                                if sentence.strip():
                                    self._release(turn, blank + sentence, not released)
                                    blank, released = "", True
                                else:
                                    blank += sentence  # white space is never sent on its own
                        else:
                            completion = event
                if completion is None:
                    raise LLMError("LLM stream ended without a completion", retryable=True,
                                   received=len(content))
                rest = buffer.flush()
                if rest.strip():
                    self._release(turn, blank + rest, not released)
            except _GuardFailed as failed:
                # the stream was cut by us; how much (hidden) output it produced is unknown
                await asyncio.shield(log_call({"content": content, "guard_problems": failed.problems,
                                               "aborted": True}))
                raise
            except asyncio.CancelledError:
                # the turn's time limit (or shutdown) interrupted this call: it is still recorded
                # and charged (06 review P2-4); `_drain` waits for the record to be written
                log_call({"content": content, "aborted": "turn_timeout"})
                raise
            except (LLMError, TimeoutError) as exc:
                if isinstance(exc, LLMError):
                    error, usage = exc, exc.usage
                    received = exc.received if exc.received is not None else (len(content) or None)
                else:  # this call's own time limit: the stream was open, its output unknown
                    error, usage, received = LLMError("LLM call timed out", retryable=True), None, None
                await asyncio.shield(log_call(
                    {"content": content, "error": str(error)[:300], "attempt": attempt + 1}, usage,
                    output_chars=received, no_response=isinstance(exc, LLMError) and received is None))
                if released or not error.retryable or attempt == self.retries:
                    raise error from None
                if (stop := turn.budget.out_of_tokens()) is not None:
                    raise _Stopped([stop]) from None  # no retry past the token cap (P2-2)
                await self._sleep(self.backoff * 2 ** attempt)
                continue
            record = completion.as_record()
            if _flooded(completion, self.budget):
                record["tool_calls_not_run"] = True  # build_context skips this completion
            await asyncio.shield(log_call(record, completion.usage, output_chars=completion.output_chars))
            return completion
        raise LLMError("unreachable", retryable=False)  # pragma: no cover

    def _city_index(self) -> CityIndex:
        """City names for the guard, from the airline's current network (rebuilt when
        locations.db changes)."""
        net = self._networks.current()
        if self._index is None or self._index[0] != net.fingerprint.id:
            self._index = (net.fingerprint.id, CityIndex(net.cities))
        return self._index[1]


def _flooded(completion: Completion, budget: AgentBudget) -> bool:
    """More tool calls in one completion than any honest step needs (P2-3)."""
    return len(completion.tool_calls) > budget.max_tool_calls_per_step * FLOOD_FACTOR
"""The four tools the LLM can call: schemas, argument validation, dispatch to the airline.

Arguments are names and ids only - never a distance or a price - so the LLM cannot pass a number
it made up. Results go back to the LLM as display values; errors are structured records too.

There is no booking tool. `Toolbox` keeps only the four bound methods it dispatches to, not the
`Airline` object, so nothing in this module can reach `Airline.book` (design P4, rule M04).
"""
from __future__ import annotations

import json
import time
from dataclasses import dataclass, replace
from typing import Any, Callable

MAX_ARG_CHARS = 200
MAX_RAW_ARGUMENTS = 2000  # the whole JSON of one call
MAX_TICKETS_FOR_LLM = 20  # the card lists them all; `count` is always the full number

    return {
        "type": "function",
        "function": {
            "name": name,
            "description": description,
            "parameters": {
                "type": "object",
                "properties": {k: {"type": "string", "description": v} for k, v in properties.items()},
                "required": required,
                "additionalProperties": False,
            },
        },
    }

DEFAULT_STATUS = "Working on it…"

@dataclass(frozen=True)
class Outcome:
    """One executed (or refused) tool call."""

    name: str
    args: Any  # parsed arguments, or the raw string when they were not valid JSON
    code: str
    payload: dict  # the JSON the LLM gets: {"code": ..., display values, names, ids}
    cards: tuple[dict, ...] = ()
    error: str | None = None  # set for calls that did not run or crashed
    latency_ms: int = 0

    def for_llm(self) -> str:
        return json.dumps(self.payload, ensure_ascii=False, separators=(",", ":"))


def slim(value: Any) -> Any:
    """Drop the 30-decimal `distance` / `price` strings where a display value sits next to them:
    the LLM only ever sees display values."""
    if isinstance(value, dict):
        return {k: slim(v) for k, v in value.items()
                if not (k in ("distance", "price") and f"{k}_display" in value)}
    if isinstance(value, (list, tuple)):
        return [slim(v) for v in value]
    return value


def _mark_nearest(item: Any) -> None:
    """On an unserved city with its `nearest` list (nearest first, as the airline ranked it): name
    the single nearest served city in `nearest_served_city`, so "use the nearest city" needs no
    comparison by the LLM. Left out when the first two are equally near at display precision
    (then the user has to choose)."""
    if not isinstance(item, dict):
        return
    near = item.get("nearest")
    if not isinstance(near, list) or not near or not all(isinstance(n, dict) for n in near):
        return
    if len(near) > 1 and near[0].get("distance_display") == near[1].get("distance_display"):
        return
    if isinstance(near[0].get("city"), str):
        item["nearest_served_city"] = near[0]["city"]


def _payload(code: str, data: dict) -> dict:
    out = {"code": code, **slim(data)}
    if code == "no_service":
        _mark_nearest(out)  # find_city
        for issue in out.get("issues") or ():  # quote_route
            _mark_nearest(issue)
    if isinstance(out.get("tickets"), list) and len(out["tickets"]) > MAX_TICKETS_FOR_LLM:
        out["tickets"] = out["tickets"][:MAX_TICKETS_FOR_LLM]  # newest first
        out["tickets_shown"] = MAX_TICKETS_FOR_LLM
    if isinstance(out.get("legs"), list) and isinstance(out.get("transfers"), list):
        # counted here so the LLM never counts (FSR: self-reported counts go wrong)
        out["leg_count"] = len(out["legs"])
        out["transfer_count"] = len(out["transfers"])
    return out


def validate(name: str, args: Any) -> tuple[dict | None, str | None]:
    """(clean arguments, None) or (None, what is wrong)."""
    spec = _SPECS[name]
    if not isinstance(args, dict):
        return None, "arguments must be a JSON object"
    unknown = sorted(set(args) - set(spec["properties"]))
    if unknown:
        return None, f"unknown argument(s): {', '.join(unknown)}; allowed: {', '.join(spec['properties'])}"
    clean: dict[str, str] = {}
    for key in spec["properties"]:
        value = args.get(key)
        if value is None or (isinstance(value, str) and not value.strip() and key not in spec["required"]):
            continue
        if not isinstance(value, str):
            return None, f"{key} must be a string"
        value = value.strip()
        if not value:
            return None, f"{key} must not be empty"
        if len(value) > MAX_ARG_CHARS:
            return None, f"{key} is too long"
        clean[key] = value
    missing = [k for k in spec["required"] if k not in clean]
    if missing:
        return None, f"missing argument(s): {', '.join(missing)}"
    return clean, None


class Toolbox:
    def __init__(self, airline: Any):
        find_city, quote_route = airline.find_city, airline.quote_route
        request_booking, list_tickets = airline.request_booking, airline.list_tickets
        self._run: dict[str, Callable[[str, str, dict], Any]] = {
            "find_city": lambda user, conv, a: find_city(a["name"]),
            "quote_route": lambda user, conv, a: quote_route(user, conv, a["origin"], a["destination"]),
            "request_booking": lambda user, conv, a: request_booking(user, conv, a["quote_id"]),
            "list_my_tickets": lambda user, conv, a: list_tickets(user, **a),
        }

    def run(self, user_id: str, conversation_id: str, name: str, raw_arguments: str) -> Outcome:
        """Validate and execute one tool call. Never raises: every failure is an Outcome the LLM
        can read and correct."""
        started = time.monotonic()

        def refused(code: str, message: str, args: Any, **extra: Any) -> Outcome:
            return Outcome(name, args, code, {"code": code, "message": message, **extra},
                           error=message, latency_ms=int((time.monotonic() - started) * 1000))

        if name not in self._run:
            return refused("unknown_tool", f"there is no tool named {name!r}", raw_arguments,
                           available=sorted(self._run))
        if not isinstance(raw_arguments, str):
            raw_arguments = ""
        if len(raw_arguments) > MAX_RAW_ARGUMENTS:
            return refused("invalid_arguments", "arguments are too long", raw_arguments[:MAX_RAW_ARGUMENTS])
        try:
            args = json.loads(raw_arguments) if raw_arguments.strip() else {}
        except (ValueError, RecursionError):
            return refused("invalid_arguments", "arguments are not valid JSON", raw_arguments)
        clean, problem = validate(name, args)
        if clean is None:
            return refused("invalid_arguments", problem or "invalid arguments", args)
        try:
            result = self._run[name](user_id, conversation_id, clean)
        except Exception as exc:  # NetworkDataError (locations.db unreadable), NotFound, bugs
            failed = refused("tool_failed", "the tool could not run; flight data may be unavailable", clean)
            return replace(failed, error=f"{type(exc).__name__}: {exc}"[:500])
        return Outcome(name, clean, result.code, _payload(result.code, result.data),
                       tuple(result.cards), latency_ms=int((time.monotonic() - started) * 1000))

"""Per-turn budgets of the agent loop (design §4.8): LLM steps, tool calls, tokens, wall time.

`AgentBudget` is the configuration (defaults below, env overrides validated by `from_env`).
`BudgetTracker` is one turn's tally. The loop asks it before every LLM step (`stop_before_step`)
and before every tool call (`admit_tool_call`); it never executes anything itself.

Token accounting is a **soft** cap and not a dollar cap:
- the provider's `usage.total_tokens` of a call is only known after the call, so the check runs
  between LLM requests: a turn that crosses the cap finishes the call in flight and starts no
  further step;
- a call is charged what the provider reported (`usage.total_tokens`, or prompt + completion
  tokens when that is larger; a total of 0 is not a report);
- a failed call that never got a response body (HTTP error status, connection / TLS failure)
  generated nothing and is charged 0 (06 review P1-2);
- any other call without a usable report is charged a conservative estimate: the request at
  `CHARS_PER_TOKEN_ESTIMATE` characters per token, plus the output at the same rate from the
  characters received (a stream cut mid-way, a completion without usage), or the whole
  `max_output_tokens_per_call` when the output is unknown (the turn's own time limit or the
  ReplyGuard cut the stream; thinking text is not seen there);
- retries are separate calls and are charged too, and a retry starts only while tokens are left.
"""
from __future__ import annotations

import json
import math
import os
from dataclasses import dataclass, fields
from typing import Any, Mapping

CHARS_PER_TOKEN_ESTIMATE = 2  # conservative: English text and JSON average about 3-4 characters per token

# name -> (environment variable, lowest, highest). Bounds keep a typo from removing a budget
# (0 or a huge number) or from making every turn fail.
_BOUNDS: dict[str, tuple[str, float, float]] = {
    "max_steps": ("AGENT_MAX_STEPS", 1, 20),
    "max_tool_calls_per_step": ("AGENT_MAX_TOOL_CALLS_PER_STEP", 1, 16),
    "max_tool_calls_per_turn": ("AGENT_MAX_TOOL_CALLS_PER_TURN", 1, 64),
    "max_output_tokens_per_call": ("AGENT_MAX_OUTPUT_TOKENS_PER_CALL", 256, 32_768),
    "max_total_tokens_per_turn": ("AGENT_MAX_TOTAL_TOKENS_PER_TURN", 1_000, 1_000_000),
    "max_turn_seconds": ("AGENT_MAX_TURN_SECONDS", 5, 600),
}


@dataclass(frozen=True)
class AgentBudget:
    max_steps: int = 6  # LLM steps (calls that complete, not counting retries) per turn
    max_tool_calls_per_step: int = 4  # executed tool calls of one completion
    max_tool_calls_per_turn: int = 12  # executed tool calls of the whole turn
    max_output_tokens_per_call: int = 4096  # the request's max_tokens (includes thinking tokens)
    # soft: provider-reported total tokens of all calls. 60k, not the first 20k: every step re-sends
    # the history, and in a 19-turn eval conversation a 3-step turn already had ~18k after step 2
    # (P1 #34, docs/iterations/06-hardening/P1/STATE.md); 60k leaves 4 steps at full history.
    max_total_tokens_per_turn: int = 60_000
    max_turn_seconds: float = 90.0  # the whole turn: LLM calls, tools, retries

    def __post_init__(self) -> None:
        for f in fields(self):
            value = getattr(self, f.name)
            whole = f.name != "max_turn_seconds"
            if isinstance(value, bool) or not isinstance(value, (int, float)) \
                    or (whole and not isinstance(value, int)) or not math.isfinite(value) or value <= 0:
                raise ValueError(f"{f.name} must be a positive {'integer' if whole else 'number'}, got {value!r}")

    def checked(self) -> "AgentBudget":
        """The same budget, also within the bounds the environment may set (startup check)."""
        for name, (var, low, high) in _BOUNDS.items():
            value = getattr(self, name)
            if not low <= value <= high:
                raise ValueError(f"{var} must be between {low:g} and {high:g}, got {value:g}")
        return self

    @classmethod
    def from_env(cls, env: Mapping[str, str] | None = None) -> "AgentBudget":
        """Defaults, overridden by the `AGENT_*` variables of `_BOUNDS`. A value that is not a
        number or is out of bounds raises ValueError (the app does not start with a bad budget)."""
        env = os.environ if env is None else env
        values: dict[str, Any] = {}
        for name, (var, low, high) in _BOUNDS.items():
            raw = (env.get(var) or "").strip().replace("_", "")
            if not raw:
                continue
            try:
                value = float(raw) if name == "max_turn_seconds" else int(raw)
            except ValueError:
                raise ValueError(f"{var} must be {'a number' if name == 'max_turn_seconds' else 'an integer'}, "
                                 f"got {raw!r}") from None
            if not low <= value <= high:  # also catches nan
                raise ValueError(f"{var} must be between {low:g} and {high:g}, got {raw!r}")
            values[name] = value
        return cls(**values).checked()


def estimate_tokens(messages: Any, max_output_tokens: int, output_chars: int | None = None) -> int:
    """Conservative charge for a call without reported usage (see the module docstring):
    the request's tokens plus the output's - from `output_chars` when known (at most
    `max_output_tokens`), else the whole `max_output_tokens`."""
    chars = len(json.dumps(messages, ensure_ascii=False, separators=(",", ":"), default=str))
    output = max_output_tokens if output_chars is None else min(
        max_output_tokens, math.ceil(max(output_chars, 0) / CHARS_PER_TOKEN_ESTIMATE))
    return math.ceil(chars / CHARS_PER_TOKEN_ESTIMATE) + output


def _count(value: Any) -> int:
    return value if isinstance(value, int) and not isinstance(value, bool) and value > 0 else 0


def reported_tokens(usage: Any) -> int | None:
    """The provider's token count of a call, or None when it reported nothing usable.
    `max(total_tokens, prompt_tokens + completion_tokens)`: a proxy that fills only the parts,
    or sends a total of 0, is not taken as "this call was free" (06 review P2-1)."""
    if not isinstance(usage, dict):
        return None
    total = max(_count(usage.get("total_tokens")),
                _count(usage.get("prompt_tokens")) + _count(usage.get("completion_tokens")))
    return total or None


class BudgetTracker:
    """One turn's use of an `AgentBudget`. Not thread-safe; owned by the turn's task."""

    def __init__(self, budget: AgentBudget):
        self.budget = budget
        self.steps = 0  # LLM steps started
        self.tool_calls = 0  # tool calls executed (refused ones not counted)
        self.refused_calls = 0
        self.tokens = 0  # reported + estimated
        self.estimated_calls = 0  # calls charged by estimate
        self._step_calls = 0

    # ---------------------------------------------------------------- LLM steps ------------
    def stop_before_step(self) -> dict | None:
        """None when another LLM step may start (and counts it), else the fallback problem."""
        if self.steps >= self.budget.max_steps:
            return {"kind": "max_steps", "value": str(self.budget.max_steps)}
        if (stop := self.out_of_tokens()) is not None:
            return stop
        self.steps += 1
        self._step_calls = 0
        return None

    def out_of_tokens(self) -> dict | None:
        """The token_budget problem once the turn's tokens are used up (checked before every
        step and before every retry), else None."""
        if self.tokens >= self.budget.max_total_tokens_per_turn:
            return {"kind": "token_budget", "value": str(self.budget.max_total_tokens_per_turn)}
        return None

    def charge(self, usage: Any, request_messages: Any, *, output_chars: int | None = None,
               no_response: bool = False) -> tuple[int, bool]:
        """Count one LLM call (completed, failed or retried); see the module docstring.
        `no_response`: the call failed before any response body was read. `output_chars`: the
        output characters received (None: unknown). Returns (tokens charged, estimated)."""
        total = reported_tokens(usage)
        if total is not None:
            self.tokens += total
            return total, False
        if no_response:
            return 0, False
        total = estimate_tokens(request_messages, self.budget.max_output_tokens_per_call, output_chars)
        self.estimated_calls += 1
        self.tokens += total
        return total, True

    # ---------------------------------------------------------------- tool calls -----------
    def admit_tool_call(self) -> str | None:
        """None when one more tool call of the current step may run (and counts it), else why
        not: "per_step" or "per_turn". A refused call is never executed."""
        if self.tool_calls >= self.budget.max_tool_calls_per_turn:
            reason = "per_turn"
        elif self._step_calls >= self.budget.max_tool_calls_per_step:
            reason = "per_step"
        else:
            self.tool_calls += 1
            self._step_calls += 1
            return None
        self.refused_calls += 1
        return reason

    @property
    def tool_calls_left(self) -> int:
        return self.budget.max_tool_calls_per_turn - self.tool_calls

    def snapshot(self) -> dict:
        """For logs and the llm_calls record (no secrets, small)."""
        return {"steps": self.steps, "tool_calls": self.tool_calls, "refused_calls": self.refused_calls,
                "tokens": self.tokens, "estimated_calls": self.estimated_calls}


def refusal(reason: str, budget: AgentBudget) -> dict:
    """The tool result of a call that was not executed (the LLM protocol needs one per call)."""
    if reason == "per_step":
        message = (f"Not executed: at most {budget.max_tool_calls_per_step} tool calls are run per step. "
                   "Call the remaining tools again in a later step if they are still needed.")
    else:
        message = (f"Not executed: this turn has used all {budget.max_tool_calls_per_turn} tool calls. "
                   "Answer with the results you already have.")
    return {"code": "tool_budget_exceeded", "message": message, "limit": reason}
